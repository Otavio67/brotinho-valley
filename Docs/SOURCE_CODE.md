```csharp
    public static class ConstructionService
    {
        public static int Cost(GameSession game, BuildingBase data)
        {
            double factor = game.State.region == "meadow" && data.animalHousing ? .75 : game.State.region == "hillside" && data.id == "fence" ? .5 : 1;
            return (int)Math.Round(data.cost * factor, MidpointRounding.AwayFromZero);
        }
        public static bool Unlocked(GameSession game, BuildingBase data) => (data.region == "" || string.IsNullOrEmpty(data.region) || data.region == game.State.region) && (LevelSystem.Level(game.State.xp) >= data.unlockLevel || data.id == "wood_workshop" && game.State.region == "forest");
        public static bool Place(GameSession game, string id, int x, int y, int rotation)
        {
            var data = game.Data.Building(id); if (data == null) return false;
            if (!Unlocked(game, data)) return game.Fail("locked");
            if (!PlacementValidator.CanPlace(game.Grid, game.Data, data, x, y, rotation)) return game.Fail("invalid_placement");
            int cost = Cost(game, data); if (!Wallet.Spend(game.State, cost)) return game.Fail("not_enough_coins");
            var building = new BuildingInstance { id = game.NextId("building"), definition = id, x = x, y = y, rotation = rotation % 4 };
            game.State.buildings.Add(building); game.LastPlacementId = building.id; game.LastPlacementCost = cost;
            game.Bus.Play(SfxEvent.Build); game.Bus.Emit(x, y, SfxEvent.Build); game.Bus.Refresh(); return true;
        }

            if (b.definition == "storage_shed" && game.State.inventory.Exists(x => x.count > game.State.Capacity - 50)) return game.Fail("storage_full");
            game.State.buildings.Remove(b); game.State.coins += game.LastPlacementCost; game.LastPlacementId = null; game.Bus.Refresh(); return true;
        }
```

```csharp
using System;
namespace Brotinho
{
    // All player and worker commands go through this service. Failed commands leave state intact.
    public sealed class GameSession
    {
        public readonly GameState State;
        public readonly GameDatabase Data;
        public readonly GridManager Grid;
        public readonly EventBus Bus;
        public string LastPlacementId;
        public long LastPlacementCost;
        public GameSession(GameState state, GameDatabase data, EventBus bus = null)
        { State = state; Data = data; Grid = new GridManager(state); Bus = bus ?? new EventBus(); OrderManager.Refresh(this); AchievementManager.RefreshDaily(state); }
        public bool Plant(string cropId, int x, int y)
        {
            var crop = Data.Crop(cropId);
            if (crop == null || Grid.State(x, y, Data) != TileState.Empty) return Fail("invalid_placement");
            if (LevelSystem.Level(State.xp) < crop.unlockLevel) return Fail("locked");
            // Forest apple saplings are free and do not consume stock, as requested.
            if (!(State.region == "forest" && cropId == "apple") && !State.Take("seed_" + cropId, 1)) return Fail("no_seeds");
            State.crops.Add(new CropInstance { id = NextId("crop"), definition = cropId, x = x, y = y });
            Bus.Play(SfxEvent.Plant); Bus.Emit(x, y, SfxEvent.Plant); Bus.Refresh(); return true;
        }
        public bool Water(int x, int y)
        {
            var crop = Grid.Crop(x, y); if (crop == null) return false;
            crop.wateredUntil = State.elapsed + 60; Bus.Play(SfxEvent.Water); Bus.Emit(x, y, SfxEvent.Water); Bus.Refresh(); return true;
        }
        public bool Remove(int x, int y)
        {
            var crop = Grid.Crop(x, y); if (crop == null) return false;
            State.crops.Remove(crop); Bus.Refresh(); return true;
        }
        public bool Fail(string key) { Bus.Notify(key); Bus.Play(SfxEvent.Error); return false; }
        public string NextId(string prefix) => prefix + "_" + State.entitySequence++;
        public void AddXp(int amount)
        {
            int old = LevelSystem.Level(State.xp); State.xp += amount;
            int current = LevelSystem.Level(State.xp);
            // A level-up milestone is every fifth level; intermediate levels grant unlocks only.
            for (int level = old + 1; level <= current; level++) if (level % 5 == 0) State.gems += 10;
            if (current > old) { Bus.Play(SfxEvent.LevelUp); Bus.Notify("level_up"); }
        }
    }
```

```csharp
    public static class SaveStateValidator

        static bool Finite(double value) => !double.IsNaN(value) && !double.IsInfinity(value) && value >= 0;

        {

            foreach (var item in s.inventory) if (item == null || string.IsNullOrEmpty(item.id) || item.count < 0) return false;
            foreach (var crop in s.crops) if (crop == null || string.IsNullOrEmpty(crop.id) || string.IsNullOrEmpty(crop.definition) || crop.x < 0 || crop.y < 0 || crop.x >= 21 || crop.y >= 21 || !Finite(crop.growth) || !Finite(crop.wateredUntil)) return false;
            foreach (var building in s.buildings) if (building == null || string.IsNullOrEmpty(building.id) || string.IsNullOrEmpty(building.definition) || building.x < 0 || building.y < 0 || building.x >= 21 || building.y >= 21 || building.queue < 0 || building.queue > 5 || building.output < 0 || !Finite(building.progress)) return false;
            foreach (var animal in s.animals) if (animal == null || string.IsNullOrEmpty(animal.definition) || animal.feed < 0 || animal.feed > 10 || animal.ready < 0 || !Finite(animal.progress)) return false;
            foreach (var worker in s.workers) if (worker == null || string.IsNullOrEmpty(worker.definition) || worker.history == null || !Finite(worker.energy) || worker.energy > 100 || !Finite(worker.cooldown) || !Enum.IsDefined(typeof(WorkerTask), worker.task) || worker.width <= 0 || worker.height <= 0 || worker.x < 0 || worker.y < 0 || worker.x + worker.width > 21 || worker.y + worker.height > 21) return false;
            foreach (var block in s.land) if (block == null || block.x < 0 || block.y < 0 || block.x + 4 > 21 || block.y + 4 > 21) return false;

        }

        {
```

```csharp
using System;
using System.IO;
using UnityEngine;
namespace Brotinho
{
    public sealed class SaveSystem
    {
        public const int CurrentVersion = 1;
        readonly string path;
        readonly Func<GameState, bool> validator;
        bool recovered;
        public bool WriteBlocked { get; private set; }
        public string LastError { get; private set; }
        public SaveSystem(string directory, Func<GameState, bool> validator = null) { path = Path.Combine(directory, "brotinho-save.json"); this.validator = validator; }
        public bool Exists => File.Exists(path) || File.Exists(path + ".bak");
        public GameState Load()
        {
            WriteBlocked = false; LastError = null; recovered = false;
            if (!Exists) return GameState.New();
            // A future-version primary is never silently replaced by an older backup.
            try { return Read(path); }
            catch (FutureSaveException) { WriteBlocked = true; LastError = "save_newer_version"; return GameState.New(); }
            catch (Exception e) when (e is IOException || e is ArgumentException || e is InvalidDataException || e is UnauthorizedAccessException) { LastError = "save_recovered"; }
            try { var state = Read(path + ".bak"); recovered = true; return state; }
            catch (FutureSaveException) { WriteBlocked = true; LastError = "save_newer_version"; return GameState.New(); }
            catch (Exception) { WriteBlocked = true; LastError = "save_corrupt"; return GameState.New(); }
        }
        GameState Read(string file)
        {
            string json = File.ReadAllText(file);
            // JsonUtility otherwise silently supplies defaults for arbitrary objects.
            if (!json.Contains("\"version\"") || !json.Contains("\"inventory\"") || !json.Contains("\"savedUtc\"")) throw new InvalidDataException("Invalid save envelope.");
            var state = JsonUtility.FromJson<GameState>(json);
            if (state == null) throw new InvalidDataException("Empty save.");
            if (state.version > CurrentVersion) throw new FutureSaveException();
            // Version 0 migrations deliberately keep the initial game schema.
            if (state.version < 0 || state.inventory == null || state.crops == null || state.buildings == null || state.animals == null || state.workers == null || state.orders == null || state.land == null || state.market == null || state.claimedAchievements == null || state.dailyClaimed == null || state.coins < 0 || state.gems < 0 || state.xp < 0 || double.IsNaN(state.elapsed) || double.IsInfinity(state.elapsed) || state.elapsed < 0 || state.tickRemainder < 0 || state.tickRemainder >= 1) throw new InvalidDataException("Invalid game state.");
            if (!SaveStateValidator.Basic(state) || validator != null && !validator(state)) throw new InvalidDataException("Invalid saved entities.");
            state.version = CurrentVersion; return state;
        }
        public bool Save(GameState state, long utcNow)
        {
            if (WriteBlocked) return false;
            try
            {
                Directory.CreateDirectory(Path.GetDirectoryName(path)); state.savedUtc = utcNow; state.version = CurrentVersion;
                string temp = path + ".tmp"; File.WriteAllText(temp, JsonUtility.ToJson(state, true));
                // Validate before rotating, keeping the previous primary as a recovery point.
                Read(temp);
                if (File.Exists(path)) File.Replace(temp, path, recovered ? null : path + ".bak"); else File.Move(temp, path);
                recovered = false;
                LastError = null; return true;
            }
            catch (Exception e) when (e is IOException || e is UnauthorizedAccessException || e is ArgumentException || e is InvalidDataException)
            { LastError = "save_failed"; return false; }
        }
        public void Reset()
        {
            // This method is called only after the in-game reset confirmation or isolated test setup.
            foreach (string suffix in new[] { "", ".bak", ".tmp" }) if (File.Exists(path + suffix)) File.Delete(path + suffix);
            WriteBlocked = false; LastError = null;
        }
        sealed class FutureSaveException : Exception { }
    }
}
```

```csharp
using System;
namespace Brotinho
{
    public static class Simulation
    {
        // Fixed one-second steps and persisted fractional remainder make chunked/offline simulation equivalent.
        // Maximum offline work is bounded to 28,800 steps, independent of system clock jumps.
        public static OfflineReport Advance(GameSession game, double seconds)
        {
            if (double.IsNaN(seconds) || double.IsInfinity(seconds) || seconds < 0) throw new ArgumentOutOfRangeException(nameof(seconds));
            var state = game.State;
            var report = new OfflineReport { seconds = seconds };
            long crops = state.harvested, products = state.produced, sales = state.sold, coins = state.coins;
            double accumulated = state.tickRemainder + seconds;
            int ticks = (int)Math.Min(int.MaxValue, Math.Floor(accumulated));
            state.tickRemainder = accumulated - ticks;
            for (int i = 0; i < ticks; i++)
            {
                AchievementManager.RefreshDaily(state);
                bool rain = WeatherSystem.At(state.elapsed) == WeatherKind.Rain;
                foreach (var crop in state.crops) CropGrowthController.Advance(crop, game.Data.Crop(crop.definition), state.region, state.elapsed, 1, rain);
                foreach (var animal in state.animals) ProductionController.Tick(game, animal);
                foreach (var building in state.buildings) ProcessingBuilding.Tick(game, building);
                foreach (var worker in state.workers) WorkerAI.Tick(game, worker);
                state.elapsed++;
                OrderManager.Refresh(game);
            }
            AchievementManager.RefreshDaily(state);
            report.harvests = state.harvested - crops; report.production = state.produced - products;
            report.sales = state.sold - sales; report.coins = state.coins - coins;
            foreach (var animal in state.animals) { if (animal.feed == 0) report.Issue("hungry_animals"); if (animal.ready >= 10) report.Issue("output_waiting"); }
            foreach (var item in state.inventory) if (item.count >= state.Capacity) report.Issue("storage_full");
            foreach (var building in state.buildings)
            {
                var data = game.Data.Building(building.definition); if (data.processSeconds <= 0) continue;
                if (building.output > 0) report.Issue("output_waiting"); else if (building.queue == 0) report.Issue("empty_input");
            }
            foreach (var worker in state.workers) if (worker.task == WorkerTask.Farm && state.Count("seed_" + worker.crop) == 0 && !(state.region == "forest" && worker.crop == "apple")) report.Issue("no_seeds");
            return report;
        }
    }
}
```

```csharp
    public static class PriceTable
    {
        public static double DailyMultiplier(string product, int day)
        {
            uint hash = 2166136261; foreach (char c in product) hash = (hash ^ c) * 16777619;
            hash = (hash ^ (uint)day) * 16777619; return (800 + hash % 401) / 1000.0;
        }
        public static double Pressure(GameState state, string product)
        {
            var sale = state.market.Find(x => x.product == product);
            if (sale == null || TimeManager.Day(state.elapsed) != TimeManager.Day(sale.lastTime)) return 0;
            return Math.Max(0, sale.pressure - (state.elapsed - sale.lastTime) / 60);
        }
        public static int Unit(GameSession game, string product, int offset = 0)
        {
            var data = game.Data.Product(product); if (data == null) return 0;
            double regional = game.State.region == "hillside" && product == "apple" ? 1.2 : 1;
            return Math.Max(1, (int)Math.Round(data.basePrice * DailyMultiplier(product, TimeManager.Day(game.State.elapsed)) * regional * Math.Max(.2, 1 - .05 * (Pressure(game.State, product) + offset)), MidpointRounding.AwayFromZero));
        }
        public static long Quote(GameSession game, string product, int amount)
        { long total = 0; for (int i = 0; i < amount; i++) total += Unit(game, product, i); return total; }
```

```csharp
using System.Collections.Generic;
using UnityEngine;
namespace Brotinho
{
    public sealed class GridManager
    {
        public const int WorldSize = 21;
        readonly GameState state;
        public GridManager(GameState state) { this.state = state; }
        public bool Inside(int x, int y) => x >= 0 && y >= 0 && x < WorldSize && y < WorldSize;
        public bool Unlocked(int x, int y)
        {
            if (!Inside(x, y)) return false;
            if (x >= 8 && x <= 12 && y >= 8 && y <= 12) return true;
            foreach (var block in state.land) if (x >= block.x && x < block.x + 4 && y >= block.y && y < block.y + 4) return true;
            return false;
        }
        public bool Water(int x, int y) => state.region == "riverside" && x == 7 || state.region != "riverside" && x == 1 && y % 3 != 0;
        public CropInstance Crop(int x, int y) { foreach (var crop in state.crops) if (crop.x == x && crop.y == y) return crop; return null; }
        public BuildingInstance Building(int x, int y, GameDatabase data)
        {
            foreach (var b in state.buildings)
            {
                var d = data.Building(b.definition); var size = PlacementValidator.Size(d, b.rotation);
                if (x >= b.x && x < b.x + size.x && y >= b.y && y < b.y + size.y) return b;
            }
            return null;
        }
        public TileState State(int x, int y, GameDatabase data)
        {
            if (!Unlocked(x, y)) return TileState.Locked;
            if (Water(x, y)) return TileState.Water;
            if (Building(x, y, data) != null) return TileState.Building;
            return Crop(x, y) != null ? TileState.Crop : TileState.Empty;
        }
        public List<Vector2Int> Path(Vector2Int start, Vector2Int destination, GameDatabase data)
        {
            var frontier = new Queue<Vector2Int>(); var cameFrom = new Dictionary<Vector2Int, Vector2Int>();
            frontier.Enqueue(start); cameFrom[start] = start;
            var dirs = new[] { Vector2Int.up, Vector2Int.right, Vector2Int.down, Vector2Int.left };
            while (frontier.Count > 0)
            {
                var current = frontier.Dequeue(); if (current == destination) break;
                foreach (var dir in dirs)
                {
                    var next = current + dir; var tile = State(next.x, next.y, data);
                    if (tile == TileState.Locked || tile == TileState.Water || tile == TileState.Building || cameFrom.ContainsKey(next)) continue;
                    cameFrom[next] = current; frontier.Enqueue(next);
                }
            }
            var result = new List<Vector2Int>(); if (!cameFrom.ContainsKey(destination)) return result;
            for (var p = destination; p != start; p = cameFrom[p]) result.Add(p);
            result.Reverse(); return result;
        }
    }
}
```

```csharp
namespace Brotinho
{
    public enum WeatherKind { Sunny, Cloudy, Rain, Storm, Dew }
    public static class WeatherSystem
    {
        public static WeatherKind At(double time)
        {
            if (time < 30) return WeatherKind.Sunny;
            uint hour = (uint)(time / TimeManager.HourSeconds);
            uint hash = hour * 747796405u + 2891336453u;
            hash = ((hash >> (int)((hash >> 28) + 4)) ^ hash) * 277803737u;
            int roll = (int)(((hash >> 22) ^ hash) % 100);
            if (TimeManager.Hour(time) < 3 && roll < 40) return WeatherKind.Dew;
            return roll < 55 ? WeatherKind.Sunny : roll < 78 ? WeatherKind.Cloudy : roll < 97 ? WeatherKind.Rain : WeatherKind.Storm;
        }
        // Storms are rare; a severe hour and the following hour form a 60-second harvest lock.
        public static bool HarvestBlocked(double time) => At(time) == WeatherKind.Storm || time >= 30 && At(time - 30) == WeatherKind.Storm;
    }
}
```

```csharp
namespace Brotinho
{
    public static class WorkerTaskQueue
    {
        public static bool InZone(WorkerInstance worker, int x, int y) => x >= worker.x && y >= worker.y && x < worker.x + worker.width && y < worker.y + worker.height;
        public static bool Execute(GameSession game, WorkerInstance worker)
        {
            if (worker.task == WorkerTask.Farm)
            {
                foreach (var c in game.State.crops)
                    if (InZone(worker, c.x, c.y) && c.Ready(game.Data.Crop(c.definition)) && !WeatherSystem.HarvestBlocked(game.State.elapsed) && game.State.CanAdd(game.Data.Crop(c.definition).product, game.Data.Crop(c.definition).yield)) return HarvestController.Harvest(game, c.x, c.y);
                for (int y = worker.y; y < worker.y + worker.height; y++) for (int x = worker.x; x < worker.x + worker.width; x++)
                    if (game.Grid.State(x, y, game.Data) == TileState.Empty && (game.State.Count("seed_" + worker.crop) > 0 || game.State.region == "forest" && worker.crop == "apple")) return game.Plant(worker.crop, x, y);
            }
            if (worker.task == WorkerTask.Animals)
            {
                foreach (var a in game.State.animals)
                {
                    var home = game.State.buildings.Find(b => b.id == a.building); if (home == null || !InZone(worker, home.x, home.y)) continue;
                    var data = game.Data.Animal(a.definition);
                    if (a.ready > 0 && game.State.CanAdd(data.product, a.ready)) return ProductionController.Collect(game, a);
                    if (a.feed < 2 && game.State.Count(data.feed) > 0) return ProductionController.Feed(game, a);
                }
            }
            if (worker.task == WorkerTask.Process)
            {
                foreach (var b in game.State.buildings)
                {
                    if (!InZone(worker, b.x, b.y)) continue;
                    var data = game.Data.Building(b.definition); if (data.processSeconds <= 0) continue;
                    if (b.output > 0 && game.State.CanAdd(data.output, b.output)) return ProcessingBuilding.Collect(game, b);
                    if (b.queue >= 5) continue;
                    bool hasInputs = true; foreach (var input in data.inputs) if (game.State.Count(input.id) < input.count) hasInputs = false;
                    if (hasInputs) return ProcessingBuilding.Enqueue(game, b);
                }
            }
            if (worker.task == WorkerTask.Sell)
            {
                if (!game.State.buildings.Exists(b => b.definition == "market_stall" && InZone(worker, b.x, b.y))) return false;
                foreach (var product in game.Data.products)
                    if (game.State.Count(product.id) > 0) return SellManager.Sell(game, product.id, 1);
            }
            return false;
        }
    }
}
```
